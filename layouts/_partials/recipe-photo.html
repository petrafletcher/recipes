{{- /* Renders a recipe's photo, in order of preference: 1. your own photo: an image named cover.* in the recipe's folder, resized to WebP; 2. the original author's photo: the `image` URL in front matter, shown from their site; 3. a placeholder. The surrounding frame carries the placeholder background, so it also shows if a remote image fails to load. Usage: partial "recipe-photo.html" (dict "page" . "size" "card"|"hero") */ -}} {{- $page := .page -}} {{- $hero := eq .size "hero" -}} {{- $class := cond $hero "hero-photo" "card-photo" -}}
{{- with $page.Resources.GetMatch "cover.*" -}} {{- $img := . -}} {{- $sizes := cond $hero (slice "720x405" "1200x675") (slice "480x300" "800x500") -}} {{- $srcset := slice -}} {{- range $sizes -}} {{- $r := $img.Fill (printf "%s webp q78 Center" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- end -}} {{- $default := $img.Fill (printf "%s webp q78 Center" (index $sizes 1)) -}} {{- else -}} {{- with $page.Params.image -}} {{- end -}} {{- end -}}
